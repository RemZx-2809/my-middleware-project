if (pathname === '/api/events' && method === 'GET') {
    res.writeHead(200, {
      'Content-Type':  'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection':    'keep-alive',
      'Access-Control-Allow-Origin': '*',
    });

    // Initial heartbeat + current store snapshot
    res.write('event: connected\ndata: {"status":"connected"}\n\n');
    res.write(`event: snapshot\ndata: ${JSON.stringify(store)}\n\n`);

    sseRegister(res);

    // Keep-alive ping every 25s (prevents proxy timeouts)
    const keepAlive = setInterval(() => {
      try { res.write(': ping\n\n'); } catch { clearInterval(keepAlive); }
    }, 25000);

    req.on('close', () => clearInterval(keepAlive));
    return; // keep connection open
  }

    /* ══════════════════════════════════════════════════════════
     PUT /api/config  — browser saves webhook secret + other settings
  ══════════════════════════════════════════════════════════ */

  if (pathname === '/api/config' && method === 'PUT') {
    if (!isAdminIpAllowed(remoteIp)) return denyAdminIp(res, remoteIp);
    if (!enforceRateLimit(`config_put:${remoteIp}`, 60000, 10)) return;

    let body;
    try { body = JSON.parse(await readBody(req)); } catch {
      return json(res, 400, { error: 'Invalid JSON' });
    }

    // Snapshot before-state (mask secret)
    const beforeSnapshot = JSON.stringify({ webhookSecretSet: !!_config.webhookSecret });

    // Merge; never expose sensitive fields back to clients
    if (typeof body.webhookSecret === 'string') _config.webhookSecret = body.webhookSecret;

    try {
      fs.writeFileSync(CONFIG_FILE, JSON.stringify(_config, null, 2), 'utf8');
    } catch (e) {
      return json(res, 500, { error: 'Could not save config', detail: e.message });
    }

    // Audit log — record config change (never log the raw secret)
    const afterSnapshot = JSON.stringify({ webhookSecretSet: !!_config.webhookSecret });
    logAudit({
      user:     req.headers['x-aegis-user'] || 'browser',
      action:   'config_update',
      filename: 'aegis.config.json',
      before:   beforeSnapshot,
      after:    afterSnapshot,
    }).catch((e) => console.warn('[AEGIS] Audit log write failed:', e.message));

    console.log('[AEGIS] Config updated from browser Settings');
    return json(res, 200, { ok: true });
  }

  /* ══════════════════════════════════════════════════════════
     GET /api/config  — browser reads config metadata (no secret exposed)
  ══════════════════════════════════════════════════════════ */
  if (pathname === '/api/config' && method === 'GET') {
    return json(res, 200, {
      webhookSecretSet: !!_config.webhookSecret,
      devMode: !_config.webhookSecret,
    });
  }

  /* ══════════════════════════════════════════════════════════
     GET /api/audit-logs  — query audit history
     Query params: limit (default 200), user, action
  ══════════════════════════════════════════════════════════ */
  if (pathname === '/api/audit-logs' && method === 'GET') {
    const limit  = parseInt(url.searchParams.get('limit')  || '200', 10);
    const user   = url.searchParams.get('user')   || '';
    const action = url.searchParams.get('action') || '';

    try {
      const logs = await getAuditLogs({ limit, user, action });
      return json(res, 200, { ok: true, count: logs.length, logs });
    } catch (e) {
      console.error('[AEGIS] Failed to read audit logs:', e.message);
      return json(res, 500, { error: 'Could not read audit logs', detail: e.message });
    }
  }

  /* ══════════════════════════════════════════════════════════
     POST /api/audit-logs  — manually create an audit entry
  ══════════════════════════════════════════════════════════ */
  if (pathname === '/api/audit-logs' && method === 'POST') {
    let body;
    try { body = JSON.parse(await readBody(req)); } catch {
      return json(res, 400, { error: 'Invalid JSON body' });
    }

    if (!body?.action) {
      return json(res, 400, { error: '"action" field is required' });
    }

    try {
      const record = await logAudit({
        user:     body.user     || req.headers['x-aegis-user'] || 'api',
        action:   body.action,
        filename: body.filename || '',
        before:   body.before   || '',
        after:    body.after    || '',
      });
      console.log(`[AEGIS] Audit entry created: ${record.action} by ${record.user}`);
      return json(res, 201, { ok: true, entry: record });
    } catch (e) {
      console.error('[AEGIS] Failed to write audit log:', e.message);
      return json(res, 500, { error: 'Could not write audit log', detail: e.message });
    }
  }

  /* ══════════════════════════════════════════════════════════
     DELETE /api/audit-logs  — clear all audit entries
  ══════════════════════════════════════════════════════════ */
  if (pathname === '/api/audit-logs' && method === 'DELETE') {
    if (!isAdminIpAllowed(remoteIp)) return denyAdminIp(res, remoteIp);
    if (!enforceRateLimit(`audit_del:${remoteIp}`, 60000, 10)) return;

    try {
      // Log the clearing action itself before wiping the file
      await logAudit({
        user:   req.headers['x-aegis-user'] || 'browser',
        action: 'audit_logs_cleared',
        before: 'audit log had entries',
        after:  'audit log wiped',
      });
      await clearAuditLogs();
      console.log('[AEGIS] Audit logs cleared');
      return json(res, 200, { ok: true, message: 'Audit logs cleared' });
    } catch (e) {
      console.error('[AEGIS] Failed to clear audit logs:', e.message);
      return json(res, 500, { error: 'Could not clear audit logs', detail: e.message });
    }
  }

  /* ══════════════════════════════════════════════════════════
     /api/rules & /rules/files — Rule file management (XML files in rules/ dir)

     GET    /api/rules or /rules/files → list all .xml files
     GET    /api/rules/:filename        → read file content
     PUT    /api/rules/:filename        → create / overwrite file
     DELETE /api/rules/:filename        → delete file
  ══════════════════════════════════════════════════════════ */
  if (pathname === '/rules/files' || pathname === '/api/rules/files') {
    if (method === 'GET') {
      try {
        const files = fs.readdirSync(RULES_DIR)
          .filter((f) => f.endsWith('.xml'))
          .map((f) => {
            const stat = fs.statSync(path.join(RULES_DIR, f));
            return { name: f, filename: f, size: stat.size, modified: stat.mtime.toISOString() };
          });
        return json(res, 200, { ok: true, status: 0, count: files.length, files, data: { affected_items: files, total_affected_items: files.length } });
      } catch (e) {
        return json(res, 500, { error: 'Could not list rules', detail: e.message });
      }
    }
  }

  if (pathname.startsWith('/api/rules') || pathname.startsWith('/rules/')) {
    // Extract optional filename after prefix
    const cleanPath = pathname.replace(/^\/api\/rules/, '').replace(/^\/rules/, '');
    const rawFile  = cleanPath.replace(/^\//, '');
    const fileName = path.basename(rawFile); // strips any directory traversal
    const filePath = rawFile && rawFile !== 'files' ? path.join(RULES_DIR, fileName) : null;

    // Security: only allow .xml files when a filename is given
    if (filePath && path.extname(fileName).toLowerCase() !== '.xml') {
      return json(res, 400, { error: 'Only .xml rule files are supported' });
    }

    /* ── GET /api/rules — list all rule files ─────────────── */
    if ((!rawFile || rawFile === 'files') && method === 'GET') {
      try {
        const files = fs.readdirSync(RULES_DIR)
          .filter((f) => f.endsWith('.xml'))
          .map((f) => {
            const stat = fs.statSync(path.join(RULES_DIR, f));
            return { name: f, filename: f, size: stat.size, modified: stat.mtime.toISOString() };
          });
        return json(res, 200, { ok: true, status: 0, count: files.length, files, data: { affected_items: files, total_affected_items: files.length } });
      } catch (e) {
        return json(res, 500, { error: 'Could not list rules', detail: e.message });
      }
    }

    /* ── GET /api/rules/:file — read a rule file ──────────── */
    if (filePath && method === 'GET') {
      if (!fs.existsSync(filePath)) {
        return json(res, 404, { error: `Rule file "${fileName}" not found` });
      }
      try {
        const content = fs.readFileSync(filePath, 'utf8');
        const body    = JSON.stringify({ ok: true, filename: fileName, content });
        res.writeHead(200, {
          'Content-Type':   'application/json',
          'Content-Length': Buffer.byteLength(body),
          'Access-Control-Allow-Origin': '*',
        });
        return res.end(body);
      } catch (e) {
        return json(res, 500, { error: 'Could not read rule file', detail: e.message });
      }
    }

    /* ── PUT /api/rules/:file — create or overwrite ──────── */
    if (filePath && method === 'PUT') {
      if (!isAdminIpAllowed(remoteIp)) return denyAdminIp(res, remoteIp);
      if (!enforceRateLimit(`rules_put:${remoteIp}`, 60000, 30)) return;

      let reqBody;
      try { reqBody = JSON.parse(await readBody(req)); } catch {
        return json(res, 400, { error: 'Invalid JSON body. Expected { "content": "<xml>..." }' });
      }

      if (typeof reqBody?.content !== 'string') {
        return json(res, 400, { error: '"content" (string) is required in request body' });
      }

      const existed = fs.existsSync(filePath);
      const before  = existed ? fs.readFileSync(filePath, 'utf8') : '';

      try {
        fs.writeFileSync(filePath, reqBody.content, 'utf8');
      } catch (e) {
        return json(res, 500, { error: 'Could not write rule file', detail: e.message });
      }

      // Audit log
      logAudit({
        user:     req.headers['x-aegis-user'] || 'api',
        action:   existed ? 'rule_updated' : 'rule_created',
        filename: fileName,
        before:   existed ? before : '(new file)',
        after:    reqBody.content,
      }).catch((e) => console.warn('[AEGIS] Audit log write failed:', e.message));

      console.log(`[AEGIS] Rule file ${existed ? 'updated' : 'created'}: ${fileName}`);
      return json(res, 200, {
        ok: true,
        action: existed ? 'updated' : 'created',
        filename: fileName,
      });
    }

    /* ── DELETE /api/rules/:file — delete a rule file ─────── */
    if (filePath && method === 'DELETE') {
      if (!isAdminIpAllowed(remoteIp)) return denyAdminIp(res, remoteIp);
      if (!enforceRateLimit(`rules_del:${remoteIp}`, 60000, 10)) return;

      if (!fs.existsSync(filePath)) {
        return json(res, 404, { error: `Rule file "${fileName}" not found` });
      }

      const before = fs.readFileSync(filePath, 'utf8');

      try {
        fs.unlinkSync(filePath);
      } catch (e) {
        return json(res, 500, { error: 'Could not delete rule file', detail: e.message });
      }

      // Audit log
      logAudit({
        user:     req.headers['x-aegis-user'] || 'api',
        action:   'rule_deleted',
        filename: fileName,
        before:   before,
        after:    '(deleted)',
      }).catch((e) => console.warn('[AEGIS] Audit log write failed:', e.message));

      console.log(`[AEGIS] Rule file deleted: ${fileName}`);
      return json(res, 200, { ok: true, message: `Rule file "${fileName}" deleted` });
    }

    // Method not supported for /api/rules
    return json(res, 405, { error: 'Method not allowed for /api/rules' });
  }

  /* ══════════════════════════════════════════════════════════
     Static file serving
  ══════════════════════════════════════════════════════════ */
  if (method !== 'GET' && method !== 'HEAD') {
    return json(res, 405, { error: 'Method not allowed' });
  }

  // Default to index.html for root
  let filePath = pathname === '/' ? '/index.html' : pathname;
  filePath = path.join(STATIC_ROOT, filePath);

  // Security: prevent path traversal
  if (!filePath.startsWith(STATIC_ROOT)) {
    return json(res, 403, { error: 'Forbidden' });
  }

  fs.stat(filePath, (err, stat) => {
    if (err || !stat.isFile()) {
      res.writeHead(404, { 'Content-Type': 'text/plain' });
      return res.end('404 Not Found');
    }

    const ext  = path.extname(filePath).toLowerCase();
    const mime = MIME[ext] || 'application/octet-stream';

    res.writeHead(200, {
      'Content-Type':   mime,
      'Content-Length': stat.size,
      'Cache-Control':  'no-cache',
    });

    if (method === 'HEAD') return res.end();
    fs.createReadStream(filePath).pipe(res);
  });
});
